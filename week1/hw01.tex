\section*{1.10}
suppose x and z are continuous random variables, and the density function of x is $f(x)$ and the density function of z is $g(z)$. 
Since x and z are independent, the joint density function of x and z is $f(x)g(z)$. Then we have
\begin{align*}
 E(x+z)&=\int \int (x+z) f(x) g(z) dx dz = \int \int x f(x) g(z) dx dz + \int \int z f(x) g(z) dx dz \\
&=\int xf(x)dx\int g(z)dz+\int f(x)dx \int z g(z)dz = E(x)+E(z)   
\end{align*}
\[
E(xz)=\int \int xzf(x)g(z) dxdz =\int xf(x) dx \int zg(z) dz=E(x)E(z)
\]
Hence, we have
\[
Var(x+z)=E((x+z)^2)-E^2(x+z)=E(x^2)+2E(x)E(z)+E(z^2)-E^2(x)-2E(x)E(z)-E^2(z)=Var(x)+Var(z)
\]
\begin{remark}
    Actually,$E(x+z)=E(x)+E(z)$ is trus for any random variables x and z, so$E(x^2+2xz+z^z)=E(x^2)+2E(x)E(z)+E(z^2)$ is also true for any random variables x and z.
\end{remark}
